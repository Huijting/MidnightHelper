"""Build Modules/ForeverClassQuestAreas.lua: per class-quest objective, the areas where its creatures spawn.

Input: cq_mobs.json (mh-research) = { textKey: { npcs: [ids], ... } }. Without it: a dry run with the Kodo only.
Spawns: Wowhead Forever tooltip JSON (map.zone = AreaTable id, map.coords per floor), via tools/gamedata.py fetch/cache.
Zone -> uiMap: DB2 UiMapAssignment (AreaID -> UiMapID) + UiMap (Type 3 = zone), Forever build.
Areas: single-linkage groups of spawn points closer than LINK map-% on the same uiMap; centre = mean; n = points.
"""
import csv
import io
import json
import os
import sys

HERE = os.path.dirname(os.path.abspath(__file__))
REPO = r"E:/World of Warcraft/_retail_/Interface/AddOns/MidnightHelper"
sys.path.insert(0, os.path.join(REPO, "tools"))
import gamedata as gd  # noqa: E402

LINK = 7.0   # map-% between two spawns of one area
MAX_AREAS = 8
MIN_POINTS = 3  # an area of 1-2 spawns is dropped when the objective has bigger ones

mobs_path = os.path.join(HERE, "cq_mobs.json")
dry = not os.path.exists(mobs_path)
if dry:
    mobs = {"FCQ_OBJ_KODO": {"npcs": [3234], "status": "GEMETEN", "src": "dry run"}}
else:
    with open(mobs_path, encoding="utf-8") as f:
        raw = {k: v for k, v in json.load(f).items() if not k.startswith("_")}
    # Sources: only what Zygor (ZY) or Wowhead Forever (WHF) say. GoldScout (GS) may NOT be used: its Data License
    # (github.com/Alkhara/GoldScout-WoW DATA_LICENSE.md v1.0, read 26 Nov 2026) forbids including "the Data, or anything
    # derived from it, in another addon". So every npc must be named by Zygor itself (##<id> in the cited guide, near
    # the cited line) or by Wowhead Forever (a WHF source); what only GS gave is dropped, also inside mixed entries.
    import re
    ZYDIR = r"E:/World of Warcraft/_forever_/Interface/AddOns/ZygorGuidesViewerForever/Guides-Forever"
    ZYFILES = {"H": "Leveling/ZygorLevelingHordeCLASSIC.lua", "A": "Leveling/ZygorLevelingAllianceCLASSIC.lua",
               "Common": "Leveling/ZygorLevelingCommonCLASSIC.lua", "H Dungeon": "Dungeons/ZygorDungeonHordeCLASSIC.lua",
               "A Dungeon": "Dungeons/ZygorDungeonAllianceCLASSIC.lua"}
    zycache = {}

    def zy_lines(key):
        if key not in zycache:
            with open(os.path.join(ZYDIR, ZYFILES[key]), encoding="utf-8", errors="replace") as f:
                zycache[key] = f.read().split("\n")
        return zycache[key]

    def zy_named(src):
        """npc ids Zygor itself names for this source: ##id within 40 lines of a cited line (whole file if none)."""
        out = set()
        for m in re.finditer(r"ZY (H Dungeon|A Dungeon|H|A|Common) CLASSIC(?::(\d+)(?:-(\d+))?)?", src):
            lines = zy_lines(m.group(1))
            if m.group(2):
                a, b = int(m.group(2)), int(m.group(3) or m.group(2))
                chunk = lines[max(0, a - 41):b + 40]
            else:
                chunk = lines
            for ln in chunk:
                out.update(int(n) for n in re.findall(r"##(\d+)", ln))
        return out

    # Use-an-item-on-a-creature steps that still need the creature's areas (26 Nov 2026, Rob: "zieke gazelles").
    raw["FCQ_OBJ_GAZELLE"] = {"q": 6129, "kind": "use", "npcs": [12296], "status": "GEMETEN",
                              "src": "ZY H CLASSIC:24730 Sickly Gazelle##12296; WHF npc 12296 tooltip (93 spawns, The Barrens)"}
    raw["FCQ_OBJ_DEER"] = {"q": 6124, "kind": "use", "npcs": [12298], "status": "GEMETEN",
                           "src": "WHF npc 12298 tooltip 'Sickly Deer' (75 spawns, Darkshore); ZY A CLASSIC:22493 names it without an id"}
    mobs, dropped, trimmed = {}, [], []
    for k, v in raw.items():
        src = v.get("src") or ""
        npcs = v.get("npcs") or []
        keep = [n for n in npcs if n in zy_named(src)]
        if not keep and "WHF" in src:
            keep = npcs  # Wowhead Forever names the creature(s) itself (dropped-by / quest text / npc page)
        if not keep:
            dropped.append(k)
            continue
        if len(keep) < len(npcs):
            trimmed.append("%s: %s -> %s" % (k, npcs, keep))
        mobs[k] = dict(v, npcs=keep)
    print("not used (no Zygor/Wowhead source for the creature, %d): %s" % (len(dropped), ", ".join(sorted(dropped))))

    # The source file in docs/ without anything from GoldScout (its Data License allows only small attributed
    # excerpts): GS parts of src and notes removed, GS-only keys moved to _skipped.
    def no_gs(s):
        parts = []
        for seg in re.split(r";\s*|\s\+\s", s or ""):
            if seg and not re.match(r"\(?GS\b", seg.strip()):
                parts.append(seg.strip())
        return "; ".join(parts)

    with open(mobs_path, encoding="utf-8") as f:
        full = json.load(f)
    clean = {"_meta": dict(full.get("_meta") or {})}
    clean["_meta"]["goldscout"] = ("removed 26 Nov 2026: GoldScout's Data License (DATA_LICENSE.md v1.0) forbids "
                                   "including its data or anything derived from it in another addon")
    clean["_meta"]["src_codes"] = "ZY = Zygor Forever guide file:line; WHF = Wowhead Forever page or tooltip"
    for k in sorted(mobs):
        v = dict(mobs[k])
        v["src"] = no_gs(v.get("src"))
        if v.get("note") and "GS" in v["note"]:
            v["note"] = "(part from GoldScout removed: licence)"
        v.pop("zones", None)
        clean[k] = v
    sk = dict(full.get("_skipped") or {})
    sk.pop("FCQ_OBJ_GAZELLE", None)
    sk.pop("FCQ_OBJ_DEER", None)
    for k in dropped:
        sk[k] = "only GoldScout named the creature (licence: not usable)"
    clean["_skipped"] = sk
    if not dry:
        gd.write_atomic(os.path.join(REPO, "docs", "forever", "CQ_MOBS_2026-11-26.json"),
                        json.dumps(clean, indent=1, ensure_ascii=False).encode("utf-8"))
        print("wrote docs/forever/CQ_MOBS_2026-11-26.json without GoldScout data")
    print("trimmed to what Zygor names (%d):\n  %s" % (len(trimmed), "\n  ".join(trimmed)))

build = gd.latest_build("wowf")


def db2(name):
    url = "https://wago.tools/db2/%s/csv?build=%s" % (name, build)
    data, _ = gd.fetch(url, os.path.join(gd.CACHE, "db2", build, name + ".csv"))
    return list(csv.DictReader(io.StringIO(data.decode("utf-8-sig"))))


uimap = {int(r["ID"]): r for r in db2("UiMap")}
assign = db2("UiMapAssignment")
print("UiMapAssignment columns:", list(assign[0].keys())[:14])
area_to_map = {}
for r in assign:
    try:
        aid, mid = int(r["AreaID"]), int(r["UiMapID"])
    except (KeyError, ValueError):
        continue
    m = uimap.get(mid)
    if not m or str(m.get("Type")) != "3":
        continue
    area_to_map.setdefault(aid, set()).add(mid)


def tooltip(npc):
    url = "https://nether.wowhead.com/forever/tooltip/npc/%d" % npc
    data, _ = gd.fetch(url, os.path.join(gd.CACHE, "tooltip", "forever", "npc", "%d.enUS.json" % npc), 3 * 86400,
                       soft404=True)
    return json.loads(data.decode("utf-8")) if data else None


def groups(points):
    left = list(points)
    out = []
    while left:
        seed = left.pop()
        grp, frontier = [seed], [seed]
        while frontier:
            p = frontier.pop()
            keep = []
            for q in left:
                if (p[0] - q[0]) ** 2 + (p[1] - q[1]) ** 2 <= LINK * LINK:
                    grp.append(q)
                    frontier.append(q)
                else:
                    keep.append(q)
            left = keep
        out.append(grp)
    return out


result, report = {}, []
for key in sorted(mobs):
    pts = {}  # uiMap -> [(x, y)]
    names = []
    for npc in mobs[key].get("npcs") or []:
        j = tooltip(int(npc))
        if not j or not j.get("name"):
            report.append("%s: npc %s unknown to Wowhead" % (key, npc))
            continue
        names.append("%s %s" % (npc, j["name"]))
        m = j.get("map") or {}
        maps = sorted(area_to_map.get(int(m.get("zone") or 0), []))
        if len(maps) != 1:
            report.append("%s: npc %s zone %s -> uiMaps %s (need exactly 1, skipped)" % (key, npc, m.get("zone"), maps))
            continue
        for _floor, cs in (m.get("coords") or {}).items():
            for c in cs:
                pts.setdefault(maps[0], []).append((float(c[0]), float(c[1])))
    areas = []
    for mid, ps in pts.items():
        for g in groups(ps):
            x = sum(p[0] for p in g) / len(g)
            y = sum(p[1] for p in g) / len(g)
            # the spawn nearest the centre (a centre can fall between two camps)
            best = min(g, key=lambda p: (p[0] - x) ** 2 + (p[1] - y) ** 2)
            areas.append((len(g), mid, round(best[0], 1), round(best[1], 1)))
    areas.sort(key=lambda a: -a[0])
    if any(a[0] >= MIN_POINTS for a in areas):
        areas = [a for a in areas if a[0] >= MIN_POINTS]
    areas = areas[:MAX_AREAS]
    if areas:
        result[key] = areas
    report.append("%s: %s -> %d area(s): %s" % (key, ", ".join(names), len(areas),
                                               "; ".join("%d pts on %d at %.1f,%.1f" % a for a in areas)))

print("\n".join(report))
lines = [
    "local _, ns = ...",
    "",
    "-- GENERATED by the build chat's cq_areas.py (26 Nov 2026) -- do not edit by hand.",
    "-- Per class-quest objective (textKey in ForeverClassQuests.lua): the areas where its creatures spawn, biggest",
    "-- first: { uiMapID, x, y, spawn points }. Creatures per objective: mh-research (docs/forever/CQ_AREAS_2026-11-26.md);",
    "-- spawn points: Wowhead Forever tooltip data; zone -> uiMap: DB2 UiMapAssignment/UiMap %s. No Polestar/GPL data." % build,
    "",
    "if not ns.IS_FOREVER then",
    "\treturn",
    "end",
    "",
    "ns.FOREVER_CQ_AREAS = {",
]
for key in sorted(result):
    body = ", ".join("{ m = %d, x = %.1f, y = %.1f, n = %d }" % (a[1], a[2], a[3], a[0]) for a in result[key])
    lines.append("\t%s = { %s }," % (key, body))
lines.append("}")
text = "\n".join(lines) + "\n"
target = os.path.join(HERE if dry else os.path.join(REPO, "Modules"), "ForeverClassQuestAreas.lua")
gd.write_atomic(target, text.encode("utf-8"))
print("%s %d objective(s) -> %s" % ("DRY RUN:" if dry else "wrote", len(result), target))
